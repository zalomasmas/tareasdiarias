  Función agregada:

  ```sql
  SET TERM ^ ;

  CREATE OR ALTER FUNCTION ESCAPE_JSON (
    TEXTO VARCHAR(8191) CHARACTER SET UTF8
  ) RETURNS VARCHAR(8191) CHARACTER SET UTF8
  AS
  BEGIN
    TEXTO = REPLACE(TEXTO, '\', '\\');
    TEXTO = REPLACE(TEXTO, '"', '\"');
    TEXTO = REPLACE(TEXTO, ASCII_CHAR(13), '');
    TEXTO = REPLACE(TEXTO, ASCII_CHAR(10), '\n');
    TEXTO = REPLACE(TEXTO, ASCII_CHAR(9), '\t');
    RETURN TEXTO;
  END
  ^

  SET TERM ; ^
  ```

  Procedure modificado (versión final, con `ESCAPE_JSON` en ambas ramas — directa y de lookup):

  ```sql
  ALTER PROCEDURE PROC_GET_JSON_GRILLA_SQL (
    COD_PROCESO_SISTEMA_CONTROL INTEGER,
    CLAVE VARCHAR(250)
  )
  RETURNS (
    "SQL" BLOB SUB_TYPE 1 SEGMENT SIZE 80
  )
  AS
  DECLARE VARIABLE COD_PROCESO_SISTEMA      INTEGER;
    DECLARE VARIABLE TABLA_ASOCIADA           VARCHAR(50);
    DECLARE VARIABLE TABLA_ABM                VARCHAR(50);
    DECLARE VARIABLE SQL_FILA                 BLOB SUB_TYPE 1 SEGMENT SIZE 80;
    DECLARE VARIABLE COL_DATASET              BLOB SUB_TYPE 1 SEGMENT SIZE 80;
    DECLARE VARIABLE COL_CAMPO_CLAVE_DATASET  VARCHAR(50);
    DECLARE VARIABLE COL_CAMPO_DESC_DATASET   VARCHAR(50);
    DECLARE VARIABLE COL_CAMPO                VARCHAR(50);
    DECLARE VARIABLE COL_CLASE                VARCHAR(50);
    DECLARE VARIABLE COL_TITULO               VARCHAR(50);
    DECLARE VARIABLE COL_ORDEN                INTEGER;
    DECLARE VARIABLE PRIMERO                  INTEGER;

    DECLARE VARIABLE SQL_LOOKUP               BLOB SUB_TYPE 1 SEGMENT SIZE 80;
    DECLARE VARIABLE CLAVE_PRIMARIA          T_NOMBRE_TABLA;
    DECLARE VARIABLE ORDEN INTEGER;
    DECLARE VARIABLE FECHA_HORA_ULT_VERSION TIMESTAMP;
    DECLARE VARIABLE COL_MACRO VARCHAR(50);
    declare variable grupos varchar(50);
    DECLARE VARIABLE CAMPO_FILTRO VARCHAR(100);
    DECLARE VARIABLE PARAMETRO VARCHAR(50);
  begin

    select list(g.cod_grupo) from grupos g where g.ORGANIZACION_PERTENECE = 1 into :grupos;

    SELECT C.COD_PROCESO_SISTEMA,COALESCE(C.TABLA_ASOCIADA,'RDB$DATABASE'),P.NOMBRE_TABLA
    FROM PROC_GET_ULTIMA_VERSION_CONTROL(:COD_PROCESO_SISTEMA_CONTROL, :grupos) C
    INNER JOIN PROCESOS_SISTEMAS P ON P.COD_PROCESO_SISTEMA = C.COD_PROCESO_SISTEMA
    INTO :COD_PROCESO_SISTEMA,:TABLA_ASOCIADA,:TABLA_ABM;

    select T.CLAVE_PRIMARIA
    from TABLAS_SISTEMAS T
    where T.NOMBRE_TABLA = :TABLA_ABM
    into :CLAVE_PRIMARIA;

    SQL_FILA = '';
    PRIMERO = 1;

    FOR
       SELECT U.ORDEN, U.FECHA_HORA_MODIFICACION  ,
              COALESCE(D.CAMPO,''),
               COALESCE(D.TITULO,''),
               COALESCE(D.CLASE,''),
               COALESCE(D.DATASET,''),
               COALESCE(D.CAMPO_CLAVE_DATASET,''),
               COALESCE(D.CAMPO_DESC_DATASET,''),
               coalesce(D.POSICION,d.orden),
               T.MACRO    ,
               COALESCE(D.CAMPO_FILTRO,'')
       FROM PROC_GET_ULT_VER_COLS_GRILLA(:COD_PROCESO_SISTEMA_CONTROL,:GRUPOS) U
       INNER JOIN DET_PROCESOS_SIST_CONTROLES D ON D.COD_PROCESO_SISTEMA_CONTROL = :COD_PROCESO_SISTEMA_CONTROL
       INNER JOIN TIPOS_CONTROLES T ON T.COD_TIPO_CONTROL = D.COD_TIPO_CONTROL
       AND D.ORDEN = U.ORDEN AND D.FECHA_HORA_MODIFICACION = U.FECHA_HORA_MODIFICACION   AND D.ACTIVO = 1 and d.grabar = 1

       ORDER BY D.POSICION, D.ORDEN
       INTO :ORDEN, :FECHA_HORA_ULT_VERSION   ,
                :COL_CAMPO,
               :COL_TITULO,
               :COL_CLASE,
               :COL_DATASET,
               :COL_CAMPO_CLAVE_DATASET,
               :COL_CAMPO_DESC_DATASET,
               :COL_ORDEN,
               :COL_MACRO ,
               :CAMPO_FILTRO

    DO
    BEGIN

      IF (COALESCE(:COL_CAMPO,'') <> '')  THEN
      BEGIN
        IF (COL_MACRO containing 'SELECT' AND COALESCE(:COL_DATASET,'') <> ''  AND COALESCE(:COL_CAMPO_CLAVE_DATASET,'') <> '' AND COALESCE(:COL_CAMPO_DESC_DATASET,'') <> '') THEN
        BEGIN

          SQL_LOOKUP = 'ESCAPE_JSON(coalesce((SELECT Q.' || :COL_CAMPO_CLAVE_DATASET || ' || '';'' || Q.' || :COL_CAMPO_DESC_DATASET || ' FROM (' || :COL_DATASET || ') Q WHERE Q.' || :COL_CAMPO_CLAVE_DATASET || '=' || 'T.' || :COL_CAMPO || '),''''))';

          FOR
             SELECT PART FROM SPLIT_STRING(:CAMPO_FILTRO,';') INTO :PARAMETRO
          DO
          BEGIN
            SQL_LOOKUP = REPLACE(:SQL_LOOKUP,'@'||:PARAMETRO,'NULL');
          END

          SQL_FILA = :SQL_FILA || IIF(:PRIMERO=0,' || '','' || ','') || '''"' || :COL_CAMPO|| '":"'' || ' || :SQL_LOOKUP || ' || ''"''';

        END
        ELSE
        BEGIN
          SQL_FILA = :SQL_FILA || IIF(:PRIMERO=0,' || '','' || ','') || '''"' || :COL_CAMPO|| '":"'' || ESCAPE_JSON(coalesce(T.' || :COL_CAMPO || ','''')) || ''"''';

        END
      END
      ELSE
      BEGIN
        SQL_FILA = :SQL_FILA || IIF(:PRIMERO=0,' || '','' || ','') || '''"COL_' || :COD_PROCESO_SISTEMA_CONTROL || '_' || :COL_ORDEN || '":""''';
      END
      PRIMERO = 0;
    END

    SQL_FILA = :SQL_FILA || IIF(:PRIMERO=0,' || '','' || ','') || '''"SEL":""''';

    SQL = 'SELECT ' || :SQL_FILA || ' as json FROM ' || :TABLA_ASOCIADA || ' T WHERE T.' || :CLAVE_PRIMARIA || '=' || '''' || COALESCE(:CLAVE,'') || '''';

    suspend;

  end
  ;
  ```
